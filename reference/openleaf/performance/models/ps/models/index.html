    class PSAeroModel(AerodynamicsModel[PSAerodynamicsData]):
    """Poll-Schuman aerodynamic model.

    Calculates drag using Reynolds number-based skin friction,
    induced drag with Oswald efficiency, and compressibility wave drag.

    Unlike simpler drag models, PS model requires atmospheric state
    (temperature, pressure) for Reynolds number calculations. When
    these are provided via drag_coef(data), the model produces more
    accurate drag coefficients.
    """

    @requires("lift_coef", "mach_number", "air_temperature", "air_pressure")
    def drag_coef(self, data: KeyedData) -> ArrayScalarLike:
        """Calculate drag coefficient from flight state data.

        Extracts lift coefficient, Mach number, and atmospheric state
        from the data dict. Unlike simpler models, PS requires
        air_temperature and air_pressure for Reynolds number effects.

        IMPORTANT: Poll-Schumann model is designed for cruise flight only.
        Non-clean configurations (takeoff, approach, landing) are not
        supported by the reference implementation.

        Args:
            data: Dictionary containing:
                - lift_coef: Lift coefficient (dimensionless)
                - mach_number: Mach number (dimensionless)
                - air_temperature: Air temperature [K]
                - air_pressure: Air pressure [Pa]

        Returns:
            Drag coefficient (dimensionless).

        Raises:
            MissingFieldError: If required fields are missing.

        Reference:
            Poll & Schumann (2021), "An estimation method for the fuel burn
            and other performance characteristics of civil transport aircraft
            in the cruise." The Aeronautical Journal, 125(1284), 257-295.
            https://doi.org/10.1017/aer.2020.62
        """
        # PS model is cruise-only - compute clean configuration drag
        return self._drag_coef_clean(
            cl=data["lift_coef"],
            mach=data["mach_number"],
            air_temperature=data["air_temperature"],
            air_pressure=data["air_pressure"],
        )

    # NOTE: PS model is cruise-only. The reference implementation (Poll & Schumann 2021,
    # pycontrails) does not define drag coefficients for non-clean configurations
    # (takeoff, approach, landing). No invented drag increments are provided here.

    def _drag_coef_clean(
        self,
        *,
        cl: ArrayScalarLike,
        mach: ArrayScalarLike,
        air_temperature: ArrayScalarLike = None,
        air_pressure: ArrayScalarLike = None,
    ) -> ArrayScalarLike:
        """Calculate drag coefficient for clean configuration.

        Note: If air_temperature and air_pressure are not provided,
        uses ISA conditions at FL350 as reference.
        """
        # Reynolds number and skin friction
        rn = reynolds_number(
            self.data.surface_area,
            mach,
            air_temperature,
            air_pressure,
        )
        c_f = skin_friction_coefficient(rn)

        # Zero-lift drag
        c_drag_0 = zero_lift_drag_coefficient(c_f, self.data.psi_0)

        # Oswald efficiency factor
        e_ls = oswald_efficiency_factor(
            c_drag_0,
            self.data.wing_aspect_ratio,
            self.data.cos_sweep,
            self.data.delta_2,
            self.data.winglets,
        )

        # Wave drag
        c_drag_w = wave_drag_coefficient(
            mach,
            cl,
            self.data.cos_sweep,
            self.data.wing_constant,
            self.data.j_1,
            self.data.j_2,
            self.data.j_3,
            self.data.x_ref,
        )

        # Total drag
        return airframe_drag_coefficient(
            c_drag_0,
            c_drag_w,
            cl,
            e_ls,
            self.data.wing_aspect_ratio,
        )

    def drag_coef_with_state(
        self,
        *,
        cl: ArrayScalarLike,
        mach: ArrayScalarLike,
        air_temperature: ArrayScalarLike,
        air_pressure: ArrayScalarLike,
    ) -> ArrayScalarLike:
        """Calculate drag coefficient using full atmospheric state.

        This is the preferred method when atmospheric conditions are known.

        Args:
            cl: Lift coefficient.
            mach: Mach number.
            air_temperature: Air temperature [K].
            air_pressure: Air pressure [Pa].

        Returns:
            Drag coefficient.
        """
        return self._drag_coef_clean(
            cl=cl,
            mach=mach,
            air_temperature=air_temperature,
            air_pressure=air_pressure,
        )

    class PSEngineModel(EngineModel[PSEngineData]):
    """Poll-Schuman engine model.

    Calculates thrust and fuel flow using propulsion efficiency
    and thrust coefficient correlations.
    """

    def __init__(self, data: PSEngineData, aero_data: PSAerodynamicsData):
        """Initialize engine model.

        Args:
            data: Engine data.
            aero_data: Aerodynamic data (needed for wing area).
        """
        super().__init__(data)
        self.aero_data = aero_data

    def thrust_coefficient(
        self,
        thrust: ArrayScalarLike,
        mach_num: ArrayScalarLike,
        air_pressure: ArrayScalarLike,
    ) -> ArrayScalarLike:
        """Calculate engine thrust coefficient.

        Args:
            thrust: Total thrust [N].
            mach_num: Mach number.
            air_pressure: Air pressure [Pa].

        Returns:
            Thrust coefficient.
        """
        return thrust / (
            0.5 * units.GAMMA * air_pressure * mach_num**2 * self.aero_data.surface_area
        )

    def thrust_from_coefficient(
        self,
        c_t: ArrayScalarLike,
        mach_num: ArrayScalarLike,
        air_pressure: ArrayScalarLike,
    ) -> ArrayScalarLike:
        """Calculate thrust from thrust coefficient.

        Args:
            c_t: Thrust coefficient.
            mach_num: Mach number.
            air_pressure: Air pressure [Pa].

        Returns:
            Thrust [N].
        """
        return c_t * 0.5 * units.GAMMA * air_pressure * mach_num**2 * self.aero_data.surface_area

    def max_thrust_coefficient(
        self,
        mach_num: ArrayScalarLike,
        air_temperature: ArrayScalarLike,
    ) -> ArrayScalarLike:
        """Calculate maximum available thrust coefficient.

        Args:
            mach_num: Mach number.
            air_temperature: Air temperature [K].

        Returns:
            Maximum available thrust coefficient (includes ``thrust_buffer``).
        """
        c_t_eta_b = thrust_coefficient_at_max_efficiency(
            mach_num, self.data.m_des, self.data.c_t_des
        )
        return max_available_thrust_coefficient(
            air_temperature,
            mach_num,
            c_t_eta_b,
            self.data.tet_mcc,
            self.data.tr_ec,
            self.data.m_ec,
            buffer=self.data.thrust_buffer,
        )

    @requires("mach_number", "air_temperature", "air_pressure")
    def thrust_max(self, data: KeyedData) -> ArrayScalarLike:
        """Calculate maximum available thrust.

        Args:
            data: Flight state containing mach_number, air_temperature, air_pressure.

        Returns:
            Maximum thrust [N].

        Raises:
            MissingFieldError: If required fields are missing.
        """
        return self._thrust_max_impl(
            mach=data["mach_number"],
            air_temperature=data["air_temperature"],
            air_pressure=data["air_pressure"],
        )

    def _thrust_max_impl(
        self,
        mach: ArrayScalarLike,
        air_temperature: ArrayScalarLike,
        air_pressure: ArrayScalarLike,
    ) -> ArrayScalarLike:
        """Calculate maximum available thrust (typed implementation).

        Args:
            mach: Mach number (dimensionless).
            air_temperature: Air temperature [K].
            air_pressure: Air pressure [Pa].

        Returns:
            Maximum thrust [N].
        """
        c_t_max = self.max_thrust_coefficient(mach, air_temperature)
        return self.thrust_from_coefficient(c_t_max, mach, air_pressure)

    @requires("mach_number")
    def thrust_min(self, data: KeyedData) -> ArrayScalarLike:
        """Calculate minimum thrust (idle).

        Args:
            data: Flight state containing mach_number.

        Returns:
            Minimum thrust [N] (approximately zero for idle).

        Raises:
            MissingFieldError: If required fields are missing.
        """
        return self._thrust_min_impl(mach=data["mach_number"])

    def _thrust_min_impl(self, mach: ArrayScalarLike) -> ArrayScalarLike:
        """Calculate minimum thrust (typed implementation).

        Args:
            mach: Mach number (used to maintain array shape).

        Returns:
            Minimum thrust [N] (approximately zero for idle).
        """
        # Idle thrust is approximately zero
        return 0.0 * mach  # Maintain array shape

    @requires("thrust", "mach_number", "air_temperature", "air_pressure", "altitude")
    def fuel_flow(self, data: KeyedData) -> ArrayScalarLike:
        """Calculate fuel flow rate.

        Args:
            data: Flight state containing thrust, mach_number, air_temperature,
                air_pressure, and altitude.

        Returns:
            Fuel flow rate [kg/s].

        Raises:
            MissingFieldError: If required fields are missing.
        """
        return self._fuel_flow_impl(
            thrust=data["thrust"],
            mach=data["mach_number"],
            air_temperature=data["air_temperature"],
            air_pressure=data["air_pressure"],
            altitude=data["altitude"],
        )

    @requires("thrust", "mach_number", "air_temperature", "air_pressure")
    def overall_propulsion_efficiency(self, data: KeyedData) -> ArrayScalarLike:
        """Calculate the overall propulsion efficiency used by fuel_flow.

        The thrust coefficient is clipped to [0, c_T,max] and the efficiency is
        floored at 0.01, exactly as in fuel_flow, so the uncorrected fuel flow is
        F V / (eta Q) with F the clipped thrust. The value is taken before the
        idle and maximum fuel flow correction (_correct_fuel_flow_impl); where
        that correction clips, F V / (fuel_flow Q) no longer equals eta.

        Args:
            data: Flight state containing thrust, mach_number, air_temperature,
                and air_pressure.

        Returns:
            Overall propulsion efficiency [-].

        Raises:
            MissingFieldError: If required fields are missing.
        """
        _, eta = self._thrust_coefficient_and_efficiency(
            thrust=data["thrust"],
            mach=data["mach_number"],
            air_temperature=data["air_temperature"],
            air_pressure=data["air_pressure"],
        )
        return eta

    def _thrust_coefficient_and_efficiency(
        self,
        thrust: ArrayScalarLike,
        mach: ArrayScalarLike,
        air_temperature: ArrayScalarLike,
        air_pressure: ArrayScalarLike,
    ) -> tuple[ArrayScalarLike, ArrayScalarLike]:
        """Calculate the clipped thrust coefficient and the efficiency at it.

        Shared by fuel_flow and overall_propulsion_efficiency so the two cannot
        diverge.

        Args:
            thrust: Thrust [N].
            mach: Mach number (dimensionless).
            air_temperature: Air temperature [K].
            air_pressure: Air pressure [Pa].

        Returns:
            Thrust coefficient clipped to [0, c_T,max] (dimensionless) and the
            overall propulsion efficiency at it, floored at 0.01 (dimensionless).
        """
        # Thrust coefficient
        c_t = self.thrust_coefficient(thrust, mach, air_pressure)

        # Thrust coefficient at max efficiency
        c_t_eta_b = thrust_coefficient_at_max_efficiency(mach, self.data.m_des, self.data.c_t_des)

        # Clamp c_t to available range
        c_t_max = self.max_thrust_coefficient(mach, air_temperature)
        c_t = np.clip(c_t, 0.0, c_t_max)

        # Overall propulsion efficiency
        eta = overall_propulsion_efficiency(
            mach,
            c_t,
            c_t_eta_b,
            self.data.m_des,
            self.data.eta_1,
            self.data.eta_2,
            engine_deterioration_factor=self.data.engine_deterioration_factor,
        )

        # Avoid division by zero
        eta = np.maximum(eta, 0.01)

        return c_t, eta

    def _fuel_flow_impl(
        self,
        thrust: ArrayScalarLike,
        mach: ArrayScalarLike,
        air_temperature: ArrayScalarLike,
        air_pressure: ArrayScalarLike,
        altitude: ArrayScalarLike,
    ) -> ArrayScalarLike:
        """Calculate fuel flow rate (typed implementation).

        Args:
            thrust: Thrust [N].
            mach: Mach number (dimensionless).
            air_temperature: Air temperature [K].
            air_pressure: Air pressure [Pa].
            altitude: Altitude [m].

        Returns:
            Fuel flow rate [kg/s].
        """
        c_t, eta = self._thrust_coefficient_and_efficiency(
            thrust=thrust,
            mach=mach,
            air_temperature=air_temperature,
            air_pressure=air_pressure,
        )

        # Fuel flow
        fuel_flow = (
            (units.GAMMA / 2)
            * (c_t * mach**3 / eta)
            * np.sqrt(units.GAMMA * units.R_AIR * air_temperature)
            * air_pressure
            * self.aero_data.surface_area
            / units.Q_FUEL
        )

        # Apply min/max corrections
        return self._correct_fuel_flow_impl(
            fuel_flow=fuel_flow,
            altitude=altitude,
            air_temperature=air_temperature,
            air_pressure=air_pressure,
            mach=mach,
        )

    def _correct_fuel_flow_impl(
        self,
        fuel_flow: ArrayScalarLike,
        altitude: ArrayScalarLike,
        air_temperature: ArrayScalarLike,
        air_pressure: ArrayScalarLike,
        mach: ArrayScalarLike,
    ) -> ArrayScalarLike:
        """Apply fuel flow corrections (typed implementation).

        Args:
            fuel_flow: Uncorrected fuel flow [kg/s].
            altitude: Altitude [m].
            air_temperature: Air temperature [K].
            air_pressure: Air pressure [Pa].
            mach: Mach number (dimensionless).

        Returns:
            Corrected fuel flow [kg/s].
        """
        altitude_ft = altitude * units.M_TO_FT

        # Minimum fuel flow (idle)
        ff_min = self._fuel_flow_idle(altitude_ft)

        # Maximum fuel flow
        theta = air_temperature / isa.TAU_SL
        delta = air_pressure / isa.P_SL
        ff_max = self._equivalent_fuel_flow_at_cruise(theta, delta, mach)

        return np.clip(fuel_flow, ff_min, ff_max)

    def _fuel_flow_idle(self, altitude_ft: ArrayScalarLike) -> ArrayScalarLike:
        """Calculate minimum fuel flow at idle.

        Args:
            altitude_ft: Altitude in feet.

        Returns:
            Idle fuel flow [kg/s].
        """
        x = altitude_ft / 10000.0
        return self.data.ff_idle_sls * (1.0 - 0.178 * x + 0.0085 * x**2)

    def _equivalent_fuel_flow_at_cruise(
        self,
        theta: ArrayScalarLike,
        delta: ArrayScalarLike,
        mach: ArrayScalarLike,
    ) -> ArrayScalarLike:
        """Calculate equivalent maximum fuel flow at cruise conditions.

        self,

        delta_isa: ArrayScalarLike,
        mach: ArrayScalarLike,

    def _excess_thrust_coefficient(
        self,
        altitude_ft: ArrayScalarLike,
        mass: ArrayScalarLike,
        delta_isa: ArrayScalarLike,
        mach: ArrayScalarLike,
    ) -> ArrayScalarLike:
        """Maximum minus required thrust coefficient at a pressure altitude.

        Positive below the thrust-limited ceiling, negative above it.

        Args:
            altitude_ft: Pressure altitude [ft].
            mass: Aircraft mass [kg].
            delta_isa: Temperature deviation from ISA [K].
            mach: Mach number.

        Returns:
            Excess thrust coefficient.
        """
        altitude_m = altitude_ft * units.FT
        air_pressure = isa.air_pressure(altitude_m)
        air_temperature = isa.air_temperature(altitude_m) + delta_isa

        # Level flight, L = m g; mass on the left keeps torch/JAX tensor types
        c_lift = (mass * units.G) / (
            0.5 * units.GAMMA * air_pressure * mach**2 * self._aerodynamics.data.surface_area
        )
        c_drag = self._aerodynamics.drag_coef_with_state(
            cl=c_lift,
            mach=mach,
            air_temperature=air_temperature,
            air_pressure=air_pressure,
        )
        true_airspeed = mach * np.sqrt(units.GAMMA * units.R_AIR * air_temperature)

        c_t_required = required_thrust_coefficient(
            c_lift, c_drag, true_airspeed, rocd=self._ceiling_climb_rate
        )
        c_t_max = self._engine.max_thrust_coefficient(mach, air_temperature)
        return c_t_max - c_t_required

    def _thrust_limited_ceiling(
        self,
        mass: ArrayScalarLike,
        delta_isa: ArrayScalarLike,
        mach: ArrayScalarLike,
    ) -> ArrayScalarLike:
        """Solve for the pressure altitude where the excess thrust is zero.

        Fixed-step bisection on ``[CEILING_FLOOR_FT, hmo]``, so the
        computation is vectorised and free of data-dependent control flow on
        every backend (CasADi turns each step into ``if_else`` nodes). The
        bisection result is piecewise constant in the inputs; one secant
        (Newton) step on the final bracket makes the root a smooth function
        of mass, temperature deviation and Mach number, so automatic
        differentiation returns the implicit-function derivative.

        If the margin holds at hmo, hmo is returned. If it fails at the
        search floor, the floor is returned: the thrust-limited ceiling then
        lies below the cruise range the PS model covers.

        Args:
            mass: Aircraft mass [kg].
            delta_isa: Temperature deviation from ISA [K].
            mach: Mach number.

        Returns:
            Thrust-limited ceiling in feet, clipped to the search bracket.
        """

        def excess(altitude_ft: ArrayScalarLike) -> ArrayScalarLike:
            return self._excess_thrust_coefficient(altitude_ft, mass, delta_isa, mach)

        excess_lo = excess(CEILING_FLOOR_FT)
        excess_hi = excess(self.data.hmo)
        # Bracket arrays take the backend, dtype and shape of the state, so
        # that np.where never promotes Python floats to a narrower dtype.
        lo = np.full_like(excess_lo, CEILING_FLOOR_FT)
        hi = np.full_like(excess_hi, self.data.hmo)

        for _ in range(CEILING_BISECTION_STEPS):
            mid = 0.5 * (lo + hi)
            excess_mid = excess(mid)
            below = excess_mid >= 0.0
            lo = np.where(below, mid, lo)
            excess_lo = np.where(below, excess_mid, excess_lo)
            hi = np.where(below, hi, mid)
            excess_hi = np.where(below, excess_hi, excess_mid)

        # Secant step inside a bracket that straddles the root. The guarded
        # denominator keeps the unselected branch finite, so gradients
        # through np.where stay finite on JAX and PyTorch.
        straddles = np.logical_and(excess_lo >= 0.0, excess_hi < 0.0)
        slope = np.where(straddles, excess_lo - excess_hi, 1.0)
        root = lo + (hi - lo) * excess_lo / slope

        return np.where(straddles, root, np.where(excess_hi >= 0.0, hi, lo))

    def _lift_limited_ceiling(
        self,
        mass: ArrayScalarLike,
        mach: ArrayScalarLike,
    ) -> ArrayScalarLike:
        """Pressure altitude at which lift reaches the maximum usable lift.

        Closed form: ``p_min = m g / (0.5 gamma M**2 S c_l_maxu(M))``.

        Args:
            mass: Aircraft mass [kg].
            mach: Mach number.

        Returns:
            Lift-limited ceiling in feet, bounded below by ``CEILING_FLOOR_FT``.
        """
        aero = self._aerodynamics.data
        c_l_maxu = max_usable_lift_coefficient(mach, self._engine.data.m_des, aero.c_l_do)
        p_min = (mass * units.G) / (0.5 * units.GAMMA * mach**2 * aero.surface_area * c_l_maxu)
        return np.maximum(isa.pressure_altitude(p_min) * units.M_TO_FT, CEILING_FLOOR_FT)

    def lift_coefficient(
        self,
        aircraft_mass: ArrayScalarLike,
        air_pressure: ArrayScalarLike,
        mach_num: ArrayScalarLike,
        climb_angle: ArrayScalarLike = 0.0,
    ) -> ArrayScalarLike:
        """Calculate lift coefficient from pressure and Mach (PS formulation).

        This uses the PS-specific formulation with air pressure and Mach.
        For model-agnostic code, use lift_coef(data) from the base class.

        Args:
            aircraft_mass: Aircraft mass [kg].
            air_pressure: Air pressure [Pa].
            mach_num: Mach number.
            climb_angle: Climb angle [degrees].

        Returns:
            Lift coefficient.
        """
        climb_angle_rad = climb_angle * np.pi / 180.0
        lift_force = aircraft_mass * units.G * np.cos(climb_angle_rad)
        denom = (
            (units.GAMMA / 2.0) * air_pressure * mach_num**2 * self.aerodynamics.data.surface_area
        )
        return lift_force / denom

        acceleration directly. For model-agnostic code, use thrust_required(data)
        from the base class which uses vertical_speed and energy_share_factor.

        Args:
            aircraft_mass: Aircraft mass [kg].
            c_lift: Lift coefficient.
            c_drag: Drag coefficient.
            dv_dt: Acceleration [m/s**2].
            climb_angle: Climb angle [degrees].

        Returns:
            Required thrust [N].

            + (aircraft_mass * units.G * np.sin(theta_rad))
            + aircraft_mass * dv_dt
        )

    def lift_coefficient(
    self,
    aircraft_mass: ArrayScalarLike,
    air_pressure: ArrayScalarLike,
    mach_num: ArrayScalarLike,
    climb_angle: ArrayScalarLike = 0.0,
) -> ArrayScalarLike:
    """Calculate lift coefficient from pressure and Mach (PS formulation).

    This uses the PS-specific formulation with air pressure and Mach.
    For model-agnostic code, use lift_coef(data) from the base class.

    Args:
        aircraft_mass: Aircraft mass [kg].
        air_pressure: Air pressure [Pa].
        mach_num: Mach number.
        climb_angle: Climb angle [degrees].

    Returns:
        Lift coefficient.
    """
    climb_angle_rad = climb_angle * np.pi / 180.0
    lift_force = aircraft_mass * units.G * np.cos(climb_angle_rad)
    denom = (
        (units.GAMMA / 2.0) * air_pressure * mach_num**2 * self.aerodynamics.data.surface_area
    )
    return lift_force / denom
