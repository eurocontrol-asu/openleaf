    numpy

    NumPy-compatible array operations supporting CasADi symbolic types.

    This module provides a drop-in replacement for NumPy that works seamlessly with both NumPy arrays and CasADi symbolic types. Functions are automatically dispatched to either NumPy or CasADi based on input types.

    Usage: import openleaf.performance.numpy as np

    Text Only
    # Works with regular NumPy arrays
x = np.array([1, 2, 3])
y = np.sqrt(x)

# Also works with CasADi symbolic types
import casadi as ca
x_sym = ca.MX.sym("x", 3)
y_sym = np.sqrt(x_sym)
    CasADi Compatibility Notes:

    The following functions work with both NumPy arrays and CasADi symbolic types: - Arithmetic: sqrt, cbrt, abs, sum, mean, prod, exp, log, log2, log10, power - Trigonometric: sin, cos, tan, arcsin, arccos, arctan, arctan2, sinh, cosh, tanh - Degree-based: sind, cosd, tand, arcsind, arccosd, arctan2d - Conditional: where, maximum, minimum, clip - Logical: logical_and, logical_or, logical_not, all, any - Array: array, concatenate, stack, reshape, roll, diag

    The following functions are NumPy-only (will NOT work with CasADi): - Sorting: sort, argsort, argmax, argmin - Statistics: std, var (use mean instead) - Array creation: linspace, arange, meshgrid, zeros, ones, full (These create NumPy arrays; CasADi arrays must be created via casadi)

    def all(a: ArrayScalarLike, axis: int | None = None) -> ArrayScalarLike:
    """Test whether all array elements along a given axis evaluate to True.

    Args:
        a: Input array or CasADi expression.
        axis: Axis or axes along which to perform the logical AND reduction.
              None means all elements.

    Returns:
        Boolean result (or array of results if axis is specified).

    Raises:
        NotImplementedError: If axis is specified for CasADi types.

    See Also:
        https://numpy.org/doc/stable/reference/generated/numpy.all.html
    """
    if is_casadi_type(a, recursive=False):
        ca = _get_casadi()
        # CasADi doesn't support axis parameter, only full reduction
        if axis is not None:
            msg = "axis parameter not supported for CasADi types in all()"
            raise NotImplementedError(msg)
        try:
            return ca.logic_all(a)
        except NotImplementedError:
            return False

    # Use Array API for NumPy/PyTorch/CuPy/JAX
    xp = get_array_namespace(a)
    return xp.all(a, axis=axis)

    def assert_equal_shape(
    arrays: list[_onp.ndarray] | dict[str, _onp.ndarray],
) -> None:
    """
    Assert that all of the given arrays are the same shape.

    Args:
        arrays: The arrays to be evaluated. Can be provided as a list or
            a dictionary of name:array pairs.

    Raises:
        ValueError: If arrays have different shapes.
    """
    names: list[str] | None = None
    array_list: list[_onp.ndarray]

    if isinstance(arrays, dict):
        names = list(arrays.keys())
        array_list = list(arrays.values())
    else:
        array_list = arrays

    def get_shape(arr) -> tuple:
        try:
            return arr.shape
        except AttributeError:
            return ()

    shape = get_shape(array_list[0])

    for arr in array_list[1:]:
        if get_shape(arr) != shape:
            if names is None:
                raise ValueError("The given arrays do not have the same shape!")
            else:
                namelist = ", ".join(names)
                raise ValueError(f"The given arrays {namelist} do not have the same shape!")

    def cbrt(x: ArrayScalarLike) -> ArrayScalarLike:
    """Calculate the cube root element-wise.

    Works with both positive and negative inputs.

    Args:
        x: Input array or scalar.

    Returns:
        Cube root of the input, element-wise.

    See Also:
        https://numpy.org/doc/stable/reference/generated/numpy.cbrt.html
    """
    if is_casadi_type(x):
        ca = _get_casadi()
        # CasADi pow doesn't handle negative bases with fractional exponents
        # Use: cbrt(x) = sign(x) * |x|^(1/3)
        return ca.sign(x) * ca.power(ca.fabs(x), 1.0 / 3.0)

    # Array API doesn't have cbrt; use sign * |x|^(1/3) for all backends
    xp = get_array_namespace(x)
    return xp.sign(x) * xp.pow(xp.abs(x), 1.0 / 3.0)

    def clip(
    x: ArrayScalarLike,
    min: ArrayScalarLike | None = None,
    max: ArrayScalarLike | None = None,
) -> ArrayScalarLike:
    """Clip (limit) the values in an array.

    Given an interval, values outside the interval are clipped to
    the interval edges. For example, if an interval of [0, 1]
    is specified, values smaller than 0 become 0, and values
    larger than 1 become 1.

    Args:
        x: Array containing elements to clip.
        min: Minimum value. If None, clipping is not performed on lower bound.
        max: Maximum value. If None, clipping is not performed on upper bound.

    Returns:
        Array with the elements of x, but where values < min are replaced
        with min, and those > max with max.

    See Also:
        https://numpy.org/doc/stable/reference/generated/numpy.clip.html
    """
    if is_casadi_type([x, min, max], recursive=True):
        ca = _get_casadi()
        result = x
        if min is not None:
            result = ca.fmax(result, min)
        if max is not None:
            result = ca.fmin(result, max)
        return result

    # Use Array API for NumPy/PyTorch/CuPy/JAX
    xp = get_array_namespace(x)
    return xp.clip(x, min, max)

    def get_array_namespace(*arrays: Any) -> Any:
    """Get the Array API namespace for the given arrays.

    Uses array-api-compat to automatically detect and return the correct
    namespace for NumPy, PyTorch, CuPy, or JAX arrays.

    Args:
        *arrays: One or more arrays to get the namespace for.

    Returns:
        The array namespace module (numpy, torch, cupy, or jax.numpy).
        Returns numpy for Python scalars (int, float, complex).

    Note:
        CasADi is NOT handled by this function - check is_casadi_type() first.

    Example:
        >>> import numpy as np
        >>> x = np.array([1, 2, 3])
        >>> xp = get_array_namespace(x)
        >>> xp.sqrt(x)
        array([1.        , 1.41421356, 1.73205081])
    """
    # Filter out Python scalars and non-array types (e.g., Enums)
    # array_namespace requires at least one proper array type
    non_scalars = []
    for a in arrays:
        # Skip Python scalars
        if isinstance(a, (int, float, complex, bool, type(None))):
            continue
        # Skip NumPy scalar types (e.g., numpy.bool_, numpy.float64).
        # These arise from operations like np.equal(enum, enum) and can
        # cause spurious namespace conflicts with JAX/PyTorch arrays.
        if isinstance(a, np.generic):
            continue
        # Skip Enum types (e.g., FlightPhase, AerodynamicConfig)
        if isinstance(a, Enum):
            continue
        # Skip strings
        if isinstance(a, str):
            continue
        non_scalars.append(a)

    if not non_scalars:
        # All inputs are scalars/enums - default to NumPy
        return Backend.NUMPY.module
    return array_api_compat.array_namespace(*non_scalars)

    One-dimensional linear interpolation, analogous to numpy.interp().

    else:
        ### Handle period argument

                raise NotImplementedError(
                    "Haven't yet implemented handling for if xp is outside the period."
                )  # Not easy to implement because casadi doesn't have a sort feature.

            x = _cas.fmod(x, period)

        ### Make sure x isn't an int
        if isinstance(x, int):
            x = float(x)

        ### Make sure that x is an iterable
        try:
            x[0]
        except TypeError:
            x = array([x], dtype=float)

        ### Make sure xp is an iterable
        xp = array(xp, dtype=float)

        ### Do the interpolation
        if is_casadi_type([x, xp], recursive=True):
            grid = [xp.shape[0]]  # size of grid, list is used since can be multi-dimensional
            cas_interp = _cas.interpolant("cs_interp", "linear", grid, 1, {"inline": True})

        if left is not None:
            f = where(x < xp[0], left, f)
        if right is not None:
            f = where(x > xp[-1], right, f)

        ### Return
        return f

    def roll(a, shift, axis: int | None = None):
    """
    Roll array elements along a given axis.

    See syntax here: https://numpy.org/doc/stable/reference/generated/numpy.roll.html
    """
    if is_casadi_type(a, recursive=False):
        if axis is None:
            a_flat = reshape(a, -1)
            result = roll(a_flat, shift, axis=0)
            return reshape(result, a.shape)
        elif isinstance(axis, int):
            shift = shift % a.shape[axis]
            if shift != 0:
                slice1 = [slice(None)] * 2
                slice1[axis] = slice(-shift, None)
                slice2 = [slice(None)] * 2
                slice2[axis] = slice(-shift)
                result = concatenate([a[tuple(slice1)], a[tuple(slice2)]], axis=axis)
            else:
                result = a
            return result
        elif isinstance(axis, tuple):
            result = a
            if not isinstance(shift, tuple):
                shift = (shift,) * len(axis)
            for ax, sh in zip(axis, shift, strict=True):
                result = roll(result, sh, ax)
            return result
        else:
            raise ValueError("'axis' must be None, an integer or a tuple of integers")

    # Use Array API for NumPy/PyTorch/CuPy/JAX
    xp = get_array_namespace(a)
    return xp.roll(a, shift, axis=axis)
